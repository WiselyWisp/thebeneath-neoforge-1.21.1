#version 150
// The Beneath: slow, seamless Essence of the Void animation.
// Only the encoded Essence marker uses this effect; other translucent
// surfaces follow their vanilla texture, tint and fog path.
#moj_import <fog.glsl>

uniform sampler2D Sampler0;
uniform vec4 ColorModulator;
uniform float FogStart;
uniform float FogEnd;
uniform vec4 FogColor;

in float vertexDistance;
in vec4 vertexColor;
in vec2 texCoord0;
in vec4 normal;
out vec4 fragColor;

const float TAU = 6.28318530718;

float hash21(vec2 p) {
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123);
}

float starField(vec2 uv, float density, float c, float s, float twinklePhase) {
    vec2 pos = uv * density;
    vec2 cell = floor(pos);
    vec2 point = fract(pos) - 0.5;
    vec2 offset = vec2(hash21(cell + 9.3), hash21(cell + 27.5)) - 0.5;
    // Staggered twinkling; the phase evolves continuously around a circle.
    float twinkle = 0.75 + 0.20 * sin(twinklePhase + hash21(cell + 11.0) * TAU);
    return (1.0 - smoothstep(0.022, 0.085, length(point - offset * 0.6)))
        * step(0.91, hash21(cell + 3.0)) * twinkle;
}

void main() {
    vec4 sampled = texture(Sampler0, texCoord0);
    // Marker channels contain sin/cos of a circular clock rather than a
    // ramp that snaps or visibly reverses at the animation boundary.
    // The base animation is 600 frames, six ticks per frame (180 seconds),
    // with Minecraft smoothly interpolating adjacent frames.
    vec3 marker = sampled.rgb * 255.0;
    bool isVoid = marker.r >= 5.0 && marker.r <= 19.0 &&
                  marker.g >= 5.0 && marker.g <= 19.0 &&
                  marker.b >= 30.0 && marker.b <= 34.0 &&
                  sampled.a > 0.98;

    vec4 color;
    if (isVoid) {
        vec2 clockVector = (marker.rg - vec2(12.0)) / 5.0;
        clockVector /= max(length(clockVector), 0.001);
        float phase = atan(clockVector.y, clockVector.x);
        // Every part of the animation is periodic, including at the wrap.
        // Different harmonic motions avoid an obvious single sliding layer.
        float p2 = 2.0 * phase;
        float p3 = 3.0 * phase;
        float p5 = 5.0 * phase;
        vec2 uv = gl_FragCoord.xy * 0.005;
        vec2 driftA = 0.080 * vec2(cos(phase), sin(phase))
                    + 0.019 * vec2(sin(p3 + 1.1), cos(p2 + 0.3));
        vec2 driftB = 0.062 * vec2(cos(p2 + 1.7), sin(p2 + 1.7))
                    + 0.015 * vec2(cos(p5), sin(p3));
        float haze = sin(uv.x * 2.1 + 0.25 * sin(phase) + 0.12 * cos(p3))
                   * cos(uv.y * 2.0 + 0.23 * cos(p2) + 0.09 * sin(p5));
        float detail = sin((uv.x + uv.y) * 1.25 + 0.16 * sin(p3 + 0.7));
        float nebula = clamp(0.50 + 0.34 * haze + 0.10 * detail, 0.0, 1.0);
        vec3 base = mix(vec3(0.008, 0.003, 0.022), vec3(0.10, 0.014, 0.17), nebula);
        float stars = starField(uv + driftA, 19.0, cos(phase), sin(phase), phase)
                    + 0.65 * starField(uv * 1.4 + driftB, 27.0,
                                       cos(p2), sin(p2), p2 + 1.25);
        color = vec4(base + stars * vec3(0.65, 0.48, 0.95), 0.96 * ColorModulator.a);
    } else {
        color = sampled * vertexColor * ColorModulator;
    }
    fragColor = linear_fog(color, vertexDistance, FogStart, FogEnd, FogColor);
}
